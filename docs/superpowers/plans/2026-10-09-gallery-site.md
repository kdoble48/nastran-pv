# Gallery Site Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Publish the docs plus an interactive 3D gallery of every nastran-pv view on GitHub Pages.

**Architecture:** Gallery assets (PNG, thumbnail, vtk.js `.vtksz` scene, JSON metadata) are made
locally by a pvpython exporter that drives the real `nastran_views`. `docs/build_gallery.py` drives
that exporter, adds a demo deck solved with MYSTRAN, zips the result and uploads it to a
`gallery-assets` release. A Sphinx site (pydata theme, MyST) builds the existing `docs/` pages.
At build time a local extension turns the downloaded assets into gallery pages with
Interactive/Static tabs that iframe trame-vtk's standalone viewer. GitHub Actions builds the site
and deploys it to Pages.

**Tech Stack:** ParaView 6.0 pvpython, trame-vtk 2.11.17 (serializers + `static_viewer.html`),
pyNastran 1.4, MYSTRAN 19.0.0, Sphinx 9.1 + pydata-sphinx-theme 0.23 + myst-parser 5.1 +
sphinx-design 0.7, Pillow, GitHub Actions (`deploy-pages`).

**Spec:** `docs/superpowers/specs/2026-10-09-gallery-site-design.md`

```python
"""Serialize a ParaView render view into a vtk.js scene (.vtksz) for a static web viewer.

Uses trame-vtk's serializers directly (no trame server): the same format PyVista's docs embed with
trame-vtk's ``static_viewer.html``. Run inside pvpython; trame-vtk is vendored into
``paraview/_vendor`` by ``install_deps.py --site``.

2D overlays (text, scalar bars, the axes triad) are not part of the scene, so :func:`overlays`
returns them as data for the web page to draw.
"""
import json
import os
import sys
import zipfile

HERE = os.path.dirname(os.path.abspath(__file__))
_VENDOR = os.path.join(HERE, "_vendor")
if os.path.isdir(_VENDOR) and _VENDOR not in sys.path:
    sys.path.append(_VENDOR)  # after ParaView's own site-packages, as the plugin does

from paraview import simple as pv  # noqa: E402

_READY = False


def _init_serializers():
    global _READY
    if _READY:
        return
    from trame_vtk.modules.vtk.serializers import initialize_serializers
    from trame_vtk.modules.vtk.serializers.registry import SERIALIZERS

    initialize_serializers()
    # ParaView 6 feeds its mappers vtkPartitionedDataSetCollection; trame-vtk registers its
    # merge-to-polydata serializer only for vtkMultiBlockDataSet and silently drops the rest.
    for name in ("vtkPartitionedDataSetCollection", "vtkPartitionedDataSet"):
        SERIALIZERS.setdefault(name, SERIALIZERS["vtkMultiBlockDataSet"])
    _READY = True


def scene_state(view, orientation_axis=True):
    """``{"hashes", "scene"}`` for ``view``, as trame-vtk's standalone (OfflineLocalView) export."""
    _init_serializers()
    from trame_vtk.modules.vtk.serializers import (SynchronizationContext, extract_array_hash,
                                                    reference_id, serialize)

    pv.Render(view)
    ctx = SynchronizationContext()
    ctx.set_ignore_last_dependencies(True)
    rw = view.GetRenderWindow()
    scene = serialize(None, rw, view.GetGlobalIDAsString(), ctx, 1)
    scene["extra"] = {"vtkRefId": reference_id(rw), "centerOfRotation": list(view.CenterOfRotation),
                      "camera": reference_id(view.GetActiveCamera())}
    if orientation_axis:
        scene["behaviors"] = {"autoOrientation": 1}
    hashes = {}
    for entry in extract_array_hash(scene):
        hashes[entry["hash"]] = dict(**entry, content=ctx.get_cached_data_array(entry["hash"], False))
    return {"hashes": hashes, "scene": scene}


def write_vtksz(view, path, orientation_axis=True):
    """Write the scene of ``view`` to ``path`` (a zip holding ``index.json``); return the state."""
    state = scene_state(view, orientation_axis)
    with zipfile.ZipFile(path, "w", zipfile.ZIP_DEFLATED) as z:
        z.writestr("index.json", json.dumps(state))
    return state


def scene_classes(state):
    """VTK class names anywhere in a scene state."""
    out, stack = set(), [state["scene"]]
    while stack:
        node = stack.pop()
        out.add(node.get("type"))
        stack.extend(node.get("dependencies", []))
    out.discard(None)
    return out


def _legend(array, lut):
    if lut.InterpretValuesAsCategories:
        ann, cols = list(lut.Annotations), list(lut.IndexedColors)
        entries = []
        for i in range(len(ann) // 2):
            rgb = cols[3 * i:3 * i + 3] or [0.5, 0.5, 0.5]
            entries.append({"value": ann[2 * i], "label": ann[2 * i + 1], "rgb": rgb})
        return {"array": array, "kind": "categorical", "entries": entries}
    pts = list(lut.RGBPoints)
    stops = [pts[i:i + 4] for i in range(0, len(pts), 4)]
    return {"array": array, "kind": "continuous", "range": [stops[0][0], stops[-1][0]], "stops": stops}


def overlays(view):
    """Visible text sources and the legends of shown scalar bars in ``view``."""
    text, legends = [], []
    for src in pv.GetSources().values():
        rep = pv.GetRepresentation(src, view)
        if rep is None or not rep.Visibility:
            continue
        if src.GetXMLName() == "TextSource":
            text.append(src.Text)
            continue
        arr = getattr(rep, "ColorArrayName", None)
        if not arr or not arr[1] or any(lg["array"] == arr[1] for lg in legends):
            continue
        lut = pv.GetColorTransferFunction(arr[1])
        if pv.GetScalarBar(lut, view).Visibility:
            legends.append(_legend(arr[1], lut))
    return {"text": text, "legends": legends}


def hide_overlays(view):
    """Hide text, scalar bars and the axes triad (for thumbnails)."""
    view.OrientationAxesVisibility = 0
    for src in pv.GetSources().values():
        rep = pv.GetRepresentation(src, view)
        if rep is None:
            continue
        if src.GetXMLName() == "TextSource":
            rep.Visibility = 0
        arr = getattr(rep, "ColorArrayName", None)
        if arr and arr[1]:
            pv.GetScalarBar(pv.GetColorTransferFunction(arr[1]), view).Visibility = 0
```

```python
"""Export nastran-pv views as web gallery assets (run with pvpython).

    pvpython paraview/export_site.py MODELS.json OUT_DIR

MODELS.json is a list of {"id", "title", "deck", "results" (OP2/H5 or null), "views", "credit"}.
For each model and view it writes <id>-<view>.png (with overlays), <id>-<view>-thumb-raw.png
(overlays hidden; cropped later by docs/build_gallery.py), <id>-<view>.vtksz (vtk.js scene) and
<id>-<view>.json (metadata). Results views get one scene and PNG per result step: -m1, -m2, ...
Views with nothing to show are skipped. Exit status 1 only when nothing was exported.
"""
import json
import math
import os
import sys

HERE = os.path.dirname(os.path.abspath(__file__))
sys.path.insert(0, HERE)

from paraview import simple as pv  # noqa: E402

import nastran_views as nv  # noqa: E402
import scene_export as se  # noqa: E402

SIZE = [1600, 1000]
VIEW_INFO = {
    "masses": ("Masses", "Mass ellipsoids from mass and inertia, colored by mass; grid-to-CG offsets; "
                         "model CG marker."),
    "bushes": ("Bushes", "CBUSH stiffness triads: K1/K2/K3 arrows along the element axes, length ~ log "
                         "stiffness, with GA–GB lines."),
    "loads": ("Loads", "Force arrows for the active LOAD set, scaled by magnitude; resultant force."),
    "spc": ("SPC", "Constrained grids colored by DOF code."),
    "rigid": ("Rigid", "RBE2/RBE3 spider legs by type."),
    "family": ("Family", "Mesh colored by element family, from the pre-processor comments."),
    "group": ("Group", "Mesh colored by top-level family group."),
    "pid": ("PID", "Mesh colored by property ID."),
    "mid": ("MID", "Mesh colored by material ID."),
    "elementtype": ("Element type", "Mesh colored by element type."),
    "thickness": ("Thickness", "Shell thickness."),
    "results": ("Results", "Deformed shape colored by displacement magnitude, one scene per mode."),
}


def new_view():
    view = pv.CreateRenderView()
    view.ViewSize = SIZE
    view.Background = [1, 1, 1]
    try:
        view.UseColorPaletteForBackground = 0
    except AttributeError:
        pass
    return view


def has_content(view):
    """True if a visible, non-text representation with data is colored by an array.

    Views that find nothing to draw show only a solid ghost mesh and an info text.
    """
    for src in pv.GetSources().values():
        rep = pv.GetRepresentation(src, view)
        if rep is None or not rep.Visibility or src.GetXMLName() == "TextSource":
            continue
        arr = getattr(rep, "ColorArrayName", None)
        if arr and arr[1] and src.GetDataInformation().GetNumberOfPoints() > 0:
            return True
    return False


def _finite(x):
    return float(x) if x is not None and math.isfinite(float(x)) else None


def export_view(reader, model, view_name, out):
    """Export one view of ``reader``; return its metadata, or None when it has nothing to show."""
    title, caption = VIEW_INFO[view_name]
    base = f"{model['id']}-{view_name}"
    steps = nv.timesteps(reader) if view_name == "results" else []
    meta, frames = None, []
    for k, t in enumerate(steps or [None], start=1):
        view = new_view()
        time = t if t is not None else 0.0
        view.ViewTime = time
        state = nv.Model(reader, time)
        nv.apply(view_name, reader=reader, view=view, model=state)
        nv.set_camera(view, "showcase")
        if not has_content(view):
            pv.Delete(view)
            return None
        stem = f"{base}-m{k}" if t is not None else base
        over = se.overlays(view)
        pv.SaveScreenshot(os.path.join(out, f"{stem}.png"), view, ImageResolution=SIZE)
        se.write_vtksz(view, os.path.join(out, f"{stem}.vtksz"))
        if meta is None:
            deck = os.path.basename(model["deck"])
            cmd = f"nastran-pv view {deck} --view {view_name}"
            if model.get("results"):
                cmd += f" --results {os.path.basename(model['results'])}"
            meta = {
                "model": model["id"], "model_title": model.get("title", model["id"]),
                "view": view_name, "title": title, "caption": caption,
                "png": f"{stem}.png", "thumb": f"{base}-thumb.png", "thumb_raw": f"{base}-thumb-raw.png",
                "scene": f"{stem}.vtksz", "scene_bytes": os.path.getsize(os.path.join(out, f"{stem}.vtksz")),
                "text": over["text"], "legends": over["legends"], "frames": frames,
                "reproduce": [cmd, f"nastran-pv render {deck} --preset {view_name} --views showcase"],
                "credit": model.get("credit", ""),
            }
            se.hide_overlays(view)
            pv.SaveScreenshot(os.path.join(out, meta["thumb_raw"]), view, ImageResolution=SIZE)
        if t is not None:
            label = state.field_strings(None, "ResultLabel")
            freq = state.field("Frequency")
            frames.append({"label": label[0] if label else f"Step {k}",
                           "frequency": _finite(freq[0]) if freq is not None and freq.size else None,
                           "scene": f"{stem}.vtksz", "png": f"{stem}.png", "text": over["text"]})
        pv.Delete(view)
    return meta


def main(argv=None):
    argv = sys.argv[1:] if argv is None else argv
    models_json, out = argv
    os.makedirs(out, exist_ok=True)
    with open(models_json, encoding="utf-8") as f:
        models = json.load(f)
    written = 0
    for model in models:
        if not os.path.exists(model["deck"]):
            print(f"skip model {model['id']}: deck not found: {model['deck']}")
            continue
        views = list(model["views"])
        results = model.get("results")
        if results and not os.path.exists(results):
            print(f"skip {model['id']}/results: results file not found: {results}")
            results, views = None, [v for v in views if v != "results"]
        model = dict(model, results=results)
        reader = nv.load_reader(os.path.abspath(model["deck"]), os.path.abspath(results) if results else None)
        for name in views:
            meta = export_view(reader, model, name, out)
            if meta is None:
                print(f"skip {model['id']}/{name}: nothing to show")
                continue
            with open(os.path.join(out, f"{model['id']}-{name}.json"), "w", encoding="utf-8") as f:
                json.dump(meta, f, indent=1, allow_nan=False)
            written += 1
            print(f"exported {model['id']}/{name} ({meta['scene_bytes'] / 1e6:.2f} MB)")
        pv.Delete(reader)
    if not written:
        print("nothing exported")
        return 1
    return 0


if __name__ == "__main__":
    sys.exit(main())

    return [m * (b * b + c * c) / 12, 0.0, m * (a * a + c * c) / 12, 0.0, 0.0, m * (a * a + b * b) / 12]

            eid += 1

def solve(bdf, mystran) -> Path:
    """Run MYSTRAN on ``bdf`` in its folder; return the OP2 it writes (case-insensitive name)."""
    bdf, mystran = Path(bdf).resolve(), Path(mystran).resolve()
    for old in bdf.parent.glob(bdf.stem + ".*"):
        if old.suffix.lower() in (".op2", ".f06"):
            old.unlink()
    subprocess.run([str(mystran), bdf.name], cwd=bdf.parent, check=False,
                   stdout=subprocess.PIPE, stderr=subprocess.STDOUT)
    for cand in bdf.parent.glob(bdf.stem + ".*"):
        if cand.suffix.lower() == ".op2" and cand.stat().st_size > 0:
            return cand
    f06 = next((c for c in bdf.parent.glob(bdf.stem + ".*") if c.suffix.lower() == ".f06"), None)

"""Generate the docs gallery assets and (optionally) upload them as a GitHub release asset.

    python docs/build_gallery.py --ucrm9 <path to run_uCRM-9_static_cruise_coarse.bdf>
        [--mystran tools/mystran-19.0.0-windows-x86_64.exe] [--pvpython PATH] [--upload]

export every view with pvpython (paraview/export_site.py); crop thumbnails; validate scenes;
write manifest.json; zip the listed files to build/gallery-assets.zip; with --upload, replace the

def crop_thumbnail(src, dst, size=(800, 500), margin=0.06, threshold=8) -> None:
    """Crop to the non-white pixels, pad to the ``size`` aspect ratio with white, resize."""
    im = Image.open(src).convert("RGB")
    diff = ImageChops.difference(im, Image.new("RGB", im.size, (255, 255, 255))).convert("L")
    x0, y0, x1, y1 = diff.point(lambda v: 255 if v > threshold else 0).getbbox() or (0, 0, *im.size)
    w, h = (x1 - x0) * (1 + 2 * margin), (y1 - y0) * (1 + 2 * margin)
    aspect = size[0] / size[1]
    w, h = (h * aspect, h) if w / h < aspect else (w, w / aspect)
    cx, cy = (x0 + x1) / 2, (y0 + y1) / 2
    box = [round(cx - w / 2), round(cy - h / 2), round(cx + w / 2), round(cy + h / 2)]
    canvas = Image.new("RGB", (box[2] - box[0], box[3] - box[1]), (255, 255, 255))
    part = im.crop((max(box[0], 0), max(box[1], 0), min(box[2], im.width), min(box[3], im.height)))
    canvas.paste(part, (max(-box[0], 0), max(-box[1], 0)))
    canvas.resize(size, Image.LANCZOS).save(dst)


def validate_scene(path) -> set[str]:
    """Class names in a .vtksz; ValueError unless it has an actor, polydata and data arrays."""
    with zipfile.ZipFile(path) as z:
        state = json.loads(z.read("index.json"))
    classes, stack = set(), [state["scene"]]
    while stack:
        node = stack.pop()
        classes.add(node.get("type"))
        stack.extend(node.get("dependencies", []))
    if not any(c and c.endswith("Actor") for c in classes):
        raise ValueError(f"{Path(path).name}: no actors in scene")
    if "vtkPolyData" not in classes or not state.get("hashes"):
        raise ValueError(f"{Path(path).name}: no geometry data in scene")
    return classes


def collect_manifest(out_dir, commit) -> dict:
    views = [json.loads(p.read_text(encoding="utf-8")) for p in Path(out_dir).glob("*.json")
             if p.name != "manifest.json"]
    rank = {m: i for i, m in enumerate(MODEL_ORDER)}
    views.sort(key=lambda v: (rank.get(v["model"], 99), VIEW_ORDER.index(v["view"])))
    now = dt.datetime.now(dt.timezone.utc).isoformat(timespec="seconds")
    return {"generated": now, "commit": commit, "views": views}


def listed_files(manifest) -> list[str]:
    names = []
    for v in manifest["views"]:
        names += [v["png"], v["thumb"], v["scene"]]
        for fr in v.get("frames", []):
            names += [fr["png"], fr["scene"]]
    return sorted(set(names))


def pack(out_dir, manifest, zip_path) -> list[str]:
    out_dir = Path(out_dir)
    (out_dir / "manifest.json").write_text(json.dumps(manifest, indent=1), encoding="utf-8")
    names = listed_files(manifest) + ["manifest.json"]
    with zipfile.ZipFile(zip_path, "w", zipfile.ZIP_DEFLATED) as z:
        for name in names:
            z.write(out_dir / name, name)
    return names


def _make_demo():
    spec = importlib.util.spec_from_file_location("make_demo", ROOT / "docs" / "models" / "make_demo.py")
    mod = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(mod)
    return mod


def _git_commit() -> str:
    r = subprocess.run(["git", "rev-parse", "--short", "HEAD"], cwd=ROOT, capture_output=True, text=True)
    return r.stdout.strip() or "unknown"


def upload(zip_path) -> None:
    if subprocess.run(["gh", "release", "view", RELEASE], cwd=ROOT, capture_output=True).returncode:
        subprocess.run(["gh", "release", "create", RELEASE, "--title", "Gallery assets", "--latest=false",
                        "--notes", "Generated by docs/build_gallery.py; downloaded by the docs workflow."],
                       cwd=ROOT, check=True)
    subprocess.run(["gh", "release", "upload", RELEASE, str(zip_path), "--clobber"], cwd=ROOT, check=True)


def main(argv=None) -> int:
    ap = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    ap.add_argument("--ucrm9", type=Path, help="uCRM-9 coarse run deck (skipped if omitted)")
    ap.add_argument("--mystran", type=Path, help="MYSTRAN executable (results view skipped if omitted)")
    ap.add_argument("--pvpython", type=Path)
    ap.add_argument("--upload", action="store_true", help=f"replace the asset on the '{RELEASE}' release")
    args = ap.parse_args(argv)

    from nastran_pv.locate import find_executable

    pvpython = args.pvpython or find_executable("pvpython")
    if OUT.exists():
        shutil.rmtree(OUT)
    OUT.mkdir(parents=True)
    md = _make_demo()
    demo_bdf = md.write_demo(OUT / "demo.bdf")
    demo_op2 = None
    if args.mystran:
        demo_op2 = md.solve(demo_bdf, args.mystran)
    else:
        print("no --mystran: demo results view skipped")
    if args.ucrm9 and not args.ucrm9.exists():
        ap.error(f"uCRM-9 deck not found: {args.ucrm9}")
    models_json = OUT / "models.json"
    models_json.write_text(json.dumps(build_models(args.ucrm9, demo_bdf, demo_op2), indent=1), encoding="utf-8")
    rc = subprocess.run(pvpython_command(pvpython, models_json, OUT), cwd=ROOT).returncode
    if rc:
        print(f"export failed (exit {rc})")
        return rc
    manifest = collect_manifest(OUT, _git_commit())
    for v in manifest["views"]:
        crop_thumbnail(OUT / v["thumb_raw"], OUT / v["thumb"])
        for scene in [v["scene"]] + [fr["scene"] for fr in v.get("frames", [])]:
            validate_scene(OUT / scene)
            size = (OUT / scene).stat().st_size
            if size > SCENE_WARN_BYTES:
                print(f"warning: {scene} is {size / 1e6:.1f} MB (budget {SCENE_WARN_BYTES / 1e6:.0f} MB)")
    names = pack(OUT, manifest, ZIP)
    print(f"{len(manifest['views'])} views, {len(names)} files -> {ZIP} ({ZIP.stat().st_size / 1e6:.1f} MB)")
    if args.upload:
        upload(ZIP)
        print(f"uploaded to release '{RELEASE}'")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())

```python
"""Sphinx extension: generate the view gallery pages from the exported assets.

On ``builder-inited`` it reads ``_static/gallery/manifest.json`` (downloaded from the
``gallery-assets`` release, or written by ``docs/build_gallery.py``), rewrites ``gallery/*.md`` and
copies trame-vtk's standalone viewer to ``_static/viewer.html``. Without a manifest the gallery is a
placeholder page. The page functions don't import Sphinx, so the tests can call them directly.
"""
from __future__ import annotations

import html
import json
import shutil
from pathlib import Path

GALLERY = "gallery"
ASSETS = "_static/gallery"
COLLAPSE_OVER = 12


def hexrgb(rgb) -> str:
    return "#" + "".join(f"{max(0, min(255, round(float(c) * 255))):02x}" for c in rgb)


def _num(x) -> str:
    return f"{float(x):.4g}"


def legend_html(leg) -> str:
    title = f'<div class="npv-legend-title">{html.escape(leg["array"])}</div>'
    if leg["kind"] == "continuous":
        lo, hi = (float(x) for x in leg["range"])
        span = (hi - lo) or 1.0
        grad = ", ".join(f"{hexrgb(s[1:4])} {100 * (float(s[0]) - lo) / span:.1f}%" for s in leg["stops"])
        return (f'<div class="npv-legend">{title}'
                f'<div class="npv-bar" style="background: linear-gradient(90deg, {grad})"></div>'
                f'<div class="npv-bar-range"><span>{_num(lo)}</span><span>{_num(hi)}</span></div></div>')
    items = "".join(f'<li><span class="npv-swatch" style="background:{hexrgb(e["rgb"])}"></span>'
                    f'{html.escape(str(e["label"]))}</li>' for e in leg["entries"])
    body = f'<ul class="npv-cats">{items}</ul>'
    if len(leg["entries"]) > COLLAPSE_OVER:
        body = f'<details><summary>{len(leg["entries"])} values</summary>{body}</details>'
    return f'<div class="npv-legend">{title}{body}</div>'


def _side(legends, text) -> str:
    summary = "\n".join(t for t in text if t).strip()
    parts = "".join(legend_html(lg) for lg in legends)
    if summary:
        parts += f'<pre class="npv-summary">{html.escape(summary)}</pre>'
    return f'```{{raw}} html\n<div class="npv-side">{parts}</div>\n```\n' if parts else ""


def _iframe(scene) -> str:
    return (f'<iframe class="npv-scene" title="3D scene" loading="lazy" '
            f'src="../_static/viewer.html?fileURL=gallery/{scene}"></iframe>')


def _mode_label(frame) -> str:
    f = frame.get("frequency")
    return f'{frame["label"]} · {f:.3g} Hz' if f is not None else frame["label"]


def view_page(meta) -> str:
    title = meta["title"]
    hint = f'Drag to rotate, scroll to zoom, right-drag to pan. Scene: {meta["scene_bytes"] / 1e6:.1f} MB.'
    if meta.get("frames"):
        tabs = "".join(f'```{{tab-item}} {_mode_label(fr)}\n{_iframe(fr["scene"])}\n\n{hint}\n```\n'
                       for fr in meta["frames"])
    else:
        tabs = f'```{{tab-item}} Interactive\n{_iframe(meta["scene"])}\n\n{hint}\n```\n'
    tabs += (f'```{{tab-item}} Static\n![{title} view of the {meta["model_title"]}]'
             f'(../{ASSETS}/{meta["png"]})\n```\n')
    reproduce = "\n".join(meta["reproduce"])
    return (f'# {title}: {meta["model_title"]}\n\n{meta["caption"]}\n\n'
            f'````{{tab-set}}\n{tabs}````\n\n{_side(meta["legends"], meta["text"])}\n'
            f'## Reproduce\n\n```bash\n{reproduce}\n```\n\n'
            f'In the ParaView GUI: open the deck with the nastran-pv reader and click **NPV {title}** '
            f'on the Macros toolbar.\n\n'
            f'```{{admonition}} Model credit\n:class: note\n{meta["credit"]}\n```\n')


def _doc(meta) -> str:
    return f'{meta["model"]}-{meta["view"]}'


def index_page(manifest) -> str:
    out = ["# View gallery\n",
           "Every one-click view, on real and demo models. Open a view to rotate and zoom it in your "
           "browser; no ParaView needed.\n"]
    models = []
    for v in manifest["views"]:
        if v["model"] not in models:
            models.append(v["model"])
    for model in models:
        views = [v for v in manifest["views"] if v["model"] == model]
        out.append(f'## {views[0]["model_title"]}\n')
        cards = [f'```{{grid-item-card}} {v["title"]}\n:link: {_doc(v)}\n:link-type: doc\n'
                 f':img-top: ../{ASSETS}/{v["thumb"]}\n\n{v["caption"]}\n```' for v in views]
        out.append("`````{grid} 1 1 2 2\n:gutter: 3\n\n" + "\n".join(cards) + "\n`````\n")
    out.append(f'Generated {manifest["generated"][:10]} from commit `{manifest["commit"]}`.\n')
    out.append("```{toctree}\n:hidden:\n\n" + "\n".join(_doc(v) for v in manifest["views"]) + "\n```\n")
    return "\n".join(out)


PLACEHOLDER = ("# View gallery\n\nThe gallery hasn't been generated yet. Maintainers: see "
               "[Regenerating the gallery](../development.md#regenerating-the-gallery).\n")


def write_pages(manifest, out_dir) -> list[Path]:
    out_dir = Path(out_dir)
    if out_dir.exists():
        shutil.rmtree(out_dir)
    out_dir.mkdir(parents=True)
    written = []
    if manifest:
        for v in manifest["views"]:
            p = out_dir / f"{_doc(v)}.md"
            p.write_text(view_page(v), encoding="utf-8")
            written.append(p)
    index = out_dir / "index.md"
    index.write_text(index_page(manifest) if manifest else PLACEHOLDER, encoding="utf-8")
    return [index] + written


def _builder_inited(app):
    src = Path(app.srcdir)
    mf = src / ASSETS / "manifest.json"
    manifest = json.loads(mf.read_text(encoding="utf-8")) if mf.exists() else None
    write_pages(manifest, src / GALLERY)
    from trame_vtk.tools.vtksz2html import HTML_VIEWER_PATH

    shutil.copy(HTML_VIEWER_PATH, src / "_static" / "viewer.html")


def setup(app):
    app.connect("builder-inited", _builder_inited)
    return {"version": "1", "parallel_read_safe": True, "parallel_write_safe": True}

concurrency:
  group: docs-${{ github.ref }}

      - uses: actions/setup-python@v5

        run: |
          mkdir -p docs/_static/gallery

            unzip -q "$RUNNER_TEMP/gallery-assets.zip" -d docs/_static/gallery
          else
            echo "::warning::gallery-assets release not found; building without the gallery"
          fi
      - run: python -m sphinx -W -b html docs docs/_build/html
      - uses: actions/upload-pages-artifact@v3
        with:
          path: docs/_build/html

  deploy:
    if: github.event_name != 'pull_request' && github.ref == 'refs/heads/main'
    needs: build
    runs-on: ubuntu-latest
    permissions:
      pages: write
      id-token: write
    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}
    steps:
      - id: deployment
        uses: actions/deploy-pages@v4
